cmake_minimum_required(VERSION 3.22.1)

project(llama_jni LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ----------------------------
# llama.cpp path configuration
# ----------------------------
# Ensure this points correctly to your submodule
set(LLAMA_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/external/llama.cpp)

if(NOT EXISTS ${LLAMA_ROOT}/CMakeLists.txt)
    message(FATAL_ERROR "llama.cpp not found at ${LLAMA_ROOT}. Check your submodule path.")
endif()

# ----------------------------
# GGML / llama.cpp Flags (Must be set BEFORE add_subdirectory)
# ----------------------------
set(GGML_VULKAN ON CACHE BOOL "Use Vulkan" FORCE)

# CRITICAL FIX: Embed shaders to prevent "Error opening file: ... .spv"
set(GGML_VULKAN_SHADER_EMBED ON CACHE BOOL "Embed Vulkan shaders" FORCE)

# Optimization flags for Android ARM64
set(GGML_USE_CPU ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

# ----------------------------
# Vulkan Header Resolution
# ----------------------------
if(ANDROID)
    # Use the Vulkan include directory passed from GitHub Actions if available,
    # otherwise fallback to the NDK sysroot.
    if(Vulkan_INCLUDE_DIR)
        include_directories(${Vulkan_INCLUDE_DIR})
    else()
        include_directories(${ANDROID_NDK}/toolchains/llvm/prebuilt/${ANDROID_HOST_TAG}/sysroot/usr/include)
    endif()
endif()

# Add llama.cpp project
add_subdirectory(${LLAMA_ROOT} ${CMAKE_BINARY_DIR}/llama)

# ----------------------------
# JNI shared library
# ----------------------------
add_library(llama_jni SHARED
    llama_jni.cpp
)

# Standardize include paths
target_include_directories(llama_jni PRIVATE
    ${LLAMA_ROOT}/include
    ${LLAMA_ROOT}/src
    ${LLAMA_ROOT}/ggml/include
    ${LLAMA_ROOT}/ggml/src
)

# Find Android-specific logs and Vulkan loader
find_library(log-lib log)
find_library(vulkan-lib vulkan)

# Link against 'llama' and 'ggml' targets generated by the subdirectory
target_link_libraries(llama_jni
    llama
    ggml
    ${log-lib}
    ${vulkan-lib}
)

# Optimization and ARM-specific features
target_compile_options(llama_jni PRIVATE
    -Wall
    -Wextra
    -fexceptions
    $<$<CONFIG:Release>:-O3 -march=armv8-a+dotprod+crypto>
)

set_target_properties(llama_jni PROPERTIES
    OUTPUT_NAME "llama_jni"
)
